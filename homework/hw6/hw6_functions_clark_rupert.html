def syllabify_diphthongs(word, vowels=hi_vowels, diphthongs=["ai", "ae", "ao", "au", "ei", "eu", "oi", "ou", "iu"]):    # default in Hawaiian, change vowels value for other languages, also: hi_vowels is defined in lin301.py
    syllable = ""
    result = []
    previous = "" # stores previous letter

    for i in word:
        if is_vowel(i, vowels) and previous + i in diphthongs:
            result[-1] += i
            previous = i
        elif is_vowel(i, vowels):
            syllable += i
            result.append(syllable) # adds string of syllable to result as list item
            syllable = ""   # clears syllable string so it is ready for next syllable
            previous = i
        else:
            syllable += i
            previous = i
    return result
